<script>
function showToast(message, isError) {
    let toast = document.getElementById('bs-toast');
    if (!toast) {
        toast = document.createElement('div');
        toast.id = 'bs-toast';
        document.body.appendChild(toast);
    }
    toast.textContent = message;
    toast.style.background = isError ? '#d9342b' : '#1a56c4';
    toast.style.opacity = '1';
    clearTimeout(toast._hideTimeout);
    toast._hideTimeout = setTimeout(() => {
        toast.style.opacity = '0';
    }, 2500);
}

async function updateCartCount() {
    try {
        const res = await fetch('/api/storefront/carts', { credentials: 'same-origin' });
        if (!res.ok) return;
        const carts = await res.json();
        const cart = carts[0];
        if (!cart) return;
        const totalQty =
            (cart.lineItems?.physicalItems || []).reduce((s, i) => s + i.quantity, 0) +
            (cart.lineItems?.digitalItems || []).reduce((s, i) => s + i.quantity, 0);

        // Default cart count selector for BigCommerce Cornerstone theme.
        // If the count doesn't update, inspect and set the actual class/attribute.
        document.querySelectorAll('.cart-quantity, [data-cart-quantity], .countPill').forEach(el => {
            el.textContent = totalQty;
            if (totalQty > 0) el.classList.add('countPill--positive');
        });
    } catch (e) {
        console.error('Cart count update error:', e);
    }
}

async function customBestSellerAddToCart(productId, sku, btn) {
    console.log('ADD FUNCTION');

    const original = btn.innerHTML;
    btn.disabled = true;
    btn.innerHTML = 'Adding...';
console.log('ADD', productId, sku);

    try {
        const formData = new FormData();
        formData.append('product_id', productId);
        formData.append('qty[]', 1);
        formData.append('sku', sku);
        const response = await fetch('/remote/v1/cart/add', {
            method: 'POST',
            body: formData,
            credentials: 'same-origin'
        });

        const result = await response.json().catch(() => ({}));

        // Handle stock/inventory or any backend-returned error
        // Error can come either at the top level or nested inside "data"
        const errMsg = result.error || result.message ||
            result.data?.error || result.data?.message ||
            (result.success === false ? 'Could not be added' : null);

        if (!response.ok || errMsg) {
            showToast(errMsg || 'Stock not available', true);
            btn.innerHTML = original;
            btn.disabled = false;
            return;
        }

        showToast('Added to cart!', false);
        btn.classList.add('added');
        btn.innerHTML = '✓ Added';
        updateCartCount();

        setTimeout(() => {
            window.location.href = 'https://undervalueappliances.ca/cart.php';
        }, 800);

    } catch (error) {
        console.error('Add To Cart Error:', error);
        showToast('Something went wrong, please try again', true);
        btn.innerHTML = original;
        btn.disabled = false;
    }
}
